db.restaurantes.aggregate([
  { $set: { promociones_vigentes: {
    $filter: {
      input: { $ifNull: ["$promociones", []] },
      as: "promocion",
      cond: {
        $and: [
          { $lte: ["$$promocion.fecha_inicio", {
            $dateTrunc: {
              date: "$$NOW", unit: "day",
              timezone: "America/Argentina/Buenos_Aires"
            }
          }] },
          { $gte: ["$$promocion.fecha_fin", {
            $dateTrunc: {
              date: "$$NOW", unit: "day",
              timezone: "America/Argentina/Buenos_Aires"
            }
          }] }
        ]
      }
    }
  } } },
  { $match: { "promociones_vigentes.0": { $exists: true } } },
  { $set: { descuento_maximo: { $max: "$promociones_vigentes.porcentaje_descuento" } } },
  { $group: {
    _id: null, maximo_global: { $max: "$descuento_maximo" },
    restaurantes: { $push: "$$ROOT" }
  } },
  { $unwind: "$restaurantes" },
  { $match: { $expr: { $eq: ["$restaurantes.descuento_maximo", "$maximo_global"] } } },
  { $project: {
    _id: 0, nombre: "$restaurantes.nombre",
    servicios_adicionales: "$restaurantes.servicios_adicionales",
    menu: "$restaurantes.menu", descuento_maximo: "$maximo_global"
  } },
  { $sort: { nombre: 1 } }
]);

db.restaurantes.aggregate([
  { $set: { eventos_con_inicio: {
    $map: {
      input: { $ifNull: ["$eventos_especiales", []] }, as: "evento",
      in: { $mergeObjects: ["$$evento", { inicio: {
        $dateFromString: {
          dateString: { $concat: [
            { $dateToString: { date: "$$evento.fecha", format: "%Y-%m-%d", timezone: "America/Argentina/Buenos_Aires" } },
            "T", "$$evento.horario", ":00"
          ] },
          timezone: "America/Argentina/Buenos_Aires"
        }
      } }] }
    }
  } } },
  { $set: { eventos_futuros: {
    $filter: { input: "$eventos_con_inicio", as: "evento", cond: { $gte: ["$$evento.inicio", "$$NOW"] } }
  } } },
  { $unwind: { path: "$eventos_futuros", preserveNullAndEmptyArrays: true } },
  { $sort: { "eventos_futuros.inicio": 1, "eventos_futuros.nombre": 1 } },
  { $group: {
    _id: "$_id", nombre: { $first: "$nombre" },
    evento_mas_proximo: { $first: { $ifNull: ["$eventos_futuros", null] } }
  } },
  { $project: { _id: 0, nombre: 1, evento_mas_proximo: 1 } },
  { $sort: { nombre: 1 } }
]);
